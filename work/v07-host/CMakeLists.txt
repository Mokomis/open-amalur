cmake_minimum_required(VERSION 3.16)
project(opem_amalur_scene_container VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID AND NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Current Android target is arm64-v8a")
endif()
add_library(amalur_blob STATIC src/amalur_blob.c)
target_include_directories(amalur_blob PUBLIC include)
add_executable(amalur-blob-inspect src/blob_inspect.c)
target_link_libraries(amalur-blob-inspect PRIVATE amalur_blob)
if(MSVC)
  target_compile_definitions(amalur_blob PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(amalur_blob PRIVATE /W4)
  target_compile_options(amalur-blob-inspect PRIVATE /W4)
else()
  target_compile_options(amalur_blob PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(amalur-blob-inspect PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ANDROID)
  target_link_options(amalur_blob INTERFACE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
option(AMALUR_BLOB_BUILD_TESTS "Build structural loader fixture tests" ON)
if(AMALUR_BLOB_BUILD_TESTS)
  add_executable(amalur-blob-test tests/blob_test.c)
  target_link_libraries(amalur-blob-test PRIVATE amalur_blob)
endif()
install(TARGETS amalur_blob amalur-blob-inspect ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/amalur_blob.h DESTINATION include)
include(android_scene.cmake)
if(ANDROID)
  set(LUA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../v07-portable/third_party/lua-5.1.5" CACHE PATH "" FORCE)
  add_subdirectory(../v07-portable portable)
  target_sources(opem_amalur_scene PRIVATE android_scripts.c)
  target_link_libraries(opem_amalur_scene PRIVATE amalur_scene_scripts amalur_scene_asset_keys amalur_scene_profile)
endif()
