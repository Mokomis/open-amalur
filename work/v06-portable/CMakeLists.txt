cmake_minimum_required(VERSION 3.16)
project(opem_amalur_script_host VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LUA_SOURCE_DIR "" CACHE PATH "Lua 5.1.5 source root, containing src/lua.h")
if(NOT EXISTS "${LUA_SOURCE_DIR}/src/lua.h")
  message(FATAL_ERROR "Set LUA_SOURCE_DIR to the unpacked Lua 5.1.5 source root")
endif()
file(STRINGS "${LUA_SOURCE_DIR}/src/lua.h" LUA_RELEASE_LINE REGEX "^#define LUA_RELEASE[ \t]+\"Lua 5\\.1\\.5\"")
if(NOT LUA_RELEASE_LINE)
  message(FATAL_ERROR "This host requires exactly Lua 5.1.5")
endif()
if(ANDROID AND NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "The current Android host target is arm64-v8a")
endif()
set(LUA_SRC
  lapi.c lcode.c ldebug.c ldo.c ldump.c lfunc.c lgc.c llex.c lmem.c lobject.c
  lopcodes.c lparser.c lstate.c lstring.c ltable.c ltm.c lundump.c lvm.c lzio.c
  lauxlib.c lbaselib.c ldblib.c liolib.c lmathlib.c loslib.c ltablib.c lstrlib.c
  loadlib.c linit.c)
list(TRANSFORM LUA_SRC PREPEND "${LUA_SOURCE_DIR}/src/")
add_library(amalur_lua51 STATIC ${LUA_SRC})
target_include_directories(amalur_lua51 PUBLIC "${LUA_SOURCE_DIR}/src")
if(NOT WIN32)
  target_link_libraries(amalur_lua51 PUBLIC m ${CMAKE_DL_LIBS})
endif()
add_library(amalur_runtime STATIC src/amalur_runtime.c)
target_include_directories(amalur_runtime PUBLIC include)
target_link_libraries(amalur_runtime PUBLIC amalur_lua51)
add_executable(amalur-script src/main.c)
target_link_libraries(amalur-script PRIVATE amalur_runtime)
if(MSVC)
  target_compile_definitions(amalur_lua51 PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(amalur_runtime PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(amalur_runtime PRIVATE /W4)
  target_compile_options(amalur-script PRIVATE /W4)
else()
  target_compile_options(amalur_runtime PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(amalur-script PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ANDROID)
  target_link_options(amalur_runtime INTERFACE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
install(TARGETS amalur_runtime amalur_lua51 amalur-script
        ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/amalur_runtime.h "${LUA_SOURCE_DIR}/src/lua.h"
        "${LUA_SOURCE_DIR}/src/luaconf.h" "${LUA_SOURCE_DIR}/src/lauxlib.h"
        "${LUA_SOURCE_DIR}/src/lualib.h" DESTINATION include)
option(AMALUR_BUILD_TESTS "Build the portable embedding API contract executable" ON)
if(AMALUR_BUILD_TESTS)
  add_executable(amalur-runtime-api-test tests/runtime_api_test.c)
  target_link_libraries(amalur-runtime-api-test PRIVATE amalur_runtime)
endif()
