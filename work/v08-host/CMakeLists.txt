cmake_minimum_required(VERSION 3.16)
project(opem_amalur_scene_container VERSION 0.8.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID AND NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Current Android target is arm64-v8a")
endif()
add_library(amalur_blob STATIC src/amalur_blob.c)
target_include_directories(amalur_blob PUBLIC include)
add_executable(amalur-blob-inspect src/blob_inspect.c)
target_link_libraries(amalur-blob-inspect PRIVATE amalur_blob)
if(MSVC)
  target_compile_definitions(amalur_blob PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(amalur_blob PRIVATE /W4)
  target_compile_options(amalur-blob-inspect PRIVATE /W4)
else()
  target_compile_options(amalur_blob PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(amalur-blob-inspect PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ANDROID)
  target_link_options(amalur_blob INTERFACE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
option(AMALUR_BLOB_BUILD_TESTS "Build structural loader fixture tests" ON)
if(AMALUR_BLOB_BUILD_TESTS)
  add_executable(amalur-blob-test tests/blob_test.c)
  target_link_libraries(amalur-blob-test PRIVATE amalur_blob)
endif()
install(TARGETS amalur_blob amalur-blob-inspect ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/amalur_blob.h DESTINATION include)
include(android_scene.cmake)
if(ANDROID)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(AMALUR_ASSET_MACRO_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/../../outputs/scene-v08/asset-macros.json" CACHE FILEPATH "Explicit asset macro manifest shared with APK packaging")
  set(AMALUR_MACRO_GENERATOR "${CMAKE_CURRENT_SOURCE_DIR}/generate_android_macro_table.py")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AMALUR_ASSET_MACRO_MANIFEST}" "${AMALUR_MACRO_GENERATOR}")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${AMALUR_MACRO_GENERATOR}"
    --manifest "${AMALUR_ASSET_MACRO_MANIFEST}"
    --output "${CMAKE_CURRENT_BINARY_DIR}/generated/android_asset_macros.inc"
    --receipt "${CMAKE_CURRENT_BINARY_DIR}/generated/android-macro-table.json"
    RESULT_VARIABLE AMALUR_MACRO_GENERATION_RESULT)
  if(NOT "${AMALUR_MACRO_GENERATION_RESULT}" STREQUAL "0")
    message(FATAL_ERROR "Android macro table generation failed; refusing stale manual descriptors")
  endif()
  target_include_directories(opem_amalur_scene PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  set(LUA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../v08-portable/third_party/lua-5.1.5" CACHE PATH "" FORCE)
  add_subdirectory(../v08-portable portable)
  target_sources(opem_amalur_scene PRIVATE android_scripts.c android_input_probe.c)
  target_link_libraries(opem_amalur_scene PRIVATE amalur_scene_scripts amalur_scene_asset_keys amalur_scene_profile_v08 amalur_scene_asset_macros_v08 amalur_scene_enums_v08)
endif()
