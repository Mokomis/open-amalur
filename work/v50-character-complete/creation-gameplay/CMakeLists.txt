cmake_minimum_required(VERSION 3.16)
project(oa_creation_gameplay_stage LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory("../../v48-opening/runtime-integration" "${CMAKE_CURRENT_BINARY_DIR}/opening-runtime" EXCLUDE_FROM_ALL)
add_library(oa_creation_gameplay STATIC oa_creation.c oa_creation_source_tables.c oa_mq01_cursor.c oa_creator_game.c oa_creation_gameplay.c)
target_include_directories(oa_creation_gameplay PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(oa_creation_gameplay PUBLIC oa_opening_runtime)
add_executable(oa-creation-gameplay-test test_creation_gameplay.c)
target_link_libraries(oa-creation-gameplay-test PRIVATE oa_creation_gameplay)
if(MSVC)
 target_compile_options(oa_creation_gameplay PRIVATE /W4 /WX)
 target_compile_definitions(oa_creation_gameplay PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(oa-creation-gameplay-test PRIVATE /W4 /WX)
 target_compile_definitions(oa-creation-gameplay-test PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
 target_compile_options(oa_creation_gameplay PRIVATE -Wall -Wextra -Werror)
 target_compile_options(oa-creation-gameplay-test PRIVATE -Wall -Wextra -Werror)
endif()
