cmake_minimum_required(VERSION 3.16)
project(oa_opening_runtime_stage LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(OA_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LUA_SOURCE_DIR "${OA_PROJECT}/work/v08-portable/third_party/lua-5.1.5" CACHE PATH "" FORCE)
set(AMALUR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${OA_PROJECT}/work/v08-portable" "${CMAKE_CURRENT_BINARY_DIR}/production-runtime" EXCLUDE_FROM_ALL)
add_library(oa_opening_runtime STATIC oa_opening_runtime.c)
target_link_libraries(oa_opening_runtime PUBLIC amalur_runtime)
target_include_directories(oa_opening_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
 target_compile_options(oa_opening_runtime PRIVATE /W4)
 target_compile_definitions(oa_opening_runtime PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
 target_compile_options(oa_opening_runtime PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(oa-opening-runtime-test oa_opening_runtime_test.c)
target_link_libraries(oa-opening-runtime-test PRIVATE oa_opening_runtime)
if(MSVC)
 target_compile_options(oa-opening-runtime-test PRIVATE /W4)
 target_compile_definitions(oa-opening-runtime-test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
