cmake_minimum_required(VERSION 3.16)
project(oa_opening_movement_stage LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../runtime-integration" "${CMAKE_CURRENT_BINARY_DIR}/opening-runtime" EXCLUDE_FROM_ALL)
add_library(oa_opening_movement STATIC oa_opening_movement.c)
target_include_directories(oa_opening_movement PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(oa_opening_movement PUBLIC oa_opening_runtime)
add_executable(oa-opening-movement-test oa_opening_movement_test.c)
target_link_libraries(oa-opening-movement-test PRIVATE oa_opening_movement)
if(MSVC)
 target_compile_options(oa_opening_movement PRIVATE /W4)
 target_compile_options(oa-opening-movement-test PRIVATE /W4)
 target_compile_definitions(oa_opening_movement PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_definitions(oa-opening-movement-test PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
 target_compile_options(oa_opening_movement PRIVATE -Wall -Wextra -Wpedantic)
endif()
