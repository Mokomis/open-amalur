cmake_minimum_required(VERSION 3.16)
project(oa_opening_gameplay_stage LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory("../runtime-integration" "${CMAKE_CURRENT_BINARY_DIR}/opening-runtime" EXCLUDE_FROM_ALL)
add_library(oa_opening_gameplay STATIC oa_opening_gameplay.c ../quest-runtime-integration/oa_mq01_cursor.c)
target_include_directories(oa_opening_gameplay PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "../quest-runtime-integration")
target_link_libraries(oa_opening_gameplay PUBLIC oa_opening_runtime)
add_executable(oa-opening-gameplay-test test_opening_gameplay.c)
target_link_libraries(oa-opening-gameplay-test PRIVATE oa_opening_gameplay)
if(MSVC)
 target_compile_options(oa_opening_gameplay PRIVATE /W4 /WX)
 target_compile_definitions(oa_opening_gameplay PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(oa-opening-gameplay-test PRIVATE /W4 /WX)
 target_compile_definitions(oa-opening-gameplay-test PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
 target_compile_options(oa_opening_gameplay PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_compile_options(oa-opening-gameplay-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
