cmake_minimum_required(VERSION 3.20)
project(open_amalur_windows_host LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "This host target is for Windows")
endif()
set(CMAKE_C_STANDARD 11)
set(HOST "${CMAKE_CURRENT_SOURCE_DIR}/../v19-host")
add_subdirectory("${HOST}" core)
set(SHARED android_scene.c android_scripts.c room_lighting.c oa_character.c oa_npc.c oa_combat.c oa_character_gl.c oa_sky.c oa_ui.c oa_menu.c oa_items.c oa_stream.c oa_post.c oa_actors.c oa_actor_ai.c oa_nav.c oa_audio.c oa_fx.c oa_adpcm.c exploration_input.c exploration_overlay.c interaction_state.c interaction_pick.c interaction_lua.c interaction_android.c opem_collision.c)
list(TRANSFORM SHARED PREPEND "${HOST}/")
add_executable(open-amalur-windows ${SHARED} desktop_platform.c desktop_input_probe.c ../v09-textures/scene_materials_v09.c ../v09-textures/scene_material_renderer_v09.c)
target_include_directories(open-amalur-windows PRIVATE include dependencies/include "${HOST}" "${HOST}/include" ../v09-textures "${HOST}/build-android-arm64/generated")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/assets/scene.bin" SCENE_HASH)
target_compile_definitions(open-amalur-windows PRIVATE _CRT_SECURE_NO_WARNINGS OA_DESKTOP=1 AMALUR_SCENE_SHA256="${SCENE_HASH}" eglSwapBuffers=oa_desktop_swap)
target_compile_options(open-amalur-windows PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/include/desktop_compat.h" /experimental:c11atomics /W3)
target_link_libraries(open-amalur-windows PRIVATE amalur_scene_mesh amalur_scene_scripts amalur_scene_asset_keys amalur_scene_profile_v08 amalur_scene_asset_macros_v08 amalur_scene_enums_v08 "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/libEGL.lib" "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/libGLESv2.lib" user32 gdi32 winmm)
add_custom_command(TARGET open-amalur-windows POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/libEGL.dll" "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/libGLESv2.dll" "$<TARGET_FILE_DIR:open-amalur-windows>")
